cmake_minimum_required(VERSION 3.16)
project(counterfeit_detection LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(verifier src/verifier.cpp)
target_include_directories(verifier PUBLIC src)
add_executable(counterfeit-check src/main.cpp)
target_link_libraries(counterfeit-check PRIVATE verifier)
add_executable(test-verifier tests/test_verifier.cpp)
target_link_libraries(test-verifier PRIVATE verifier)
enable_testing()
add_test(NAME verifier_tests COMMAND test-verifier)
